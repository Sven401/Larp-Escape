cmake_minimum_required(VERSION 3.24)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../scripts/check_idf_sources.py" "$ENV{IDF_PATH}" COMMAND_ERROR_IS_FATAL ANY)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../idf/components")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(COMPONENTS main)
set(PROJECT_VER "native-ble-qualification")
project(larp_native_ble)
if(NOT "$ENV{IDF_VERSION}" STREQUAL "6.1.0" OR NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "Only the pinned ESP-IDF 6.1.0 / ESP32 pair is qualified")
endif()
